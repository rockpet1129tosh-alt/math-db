\documentclass[border=1pt]{standalone}
\usepackage{tikz}

\begin{document}
\begin{tikzpicture}[x=2.9mm, y=2.9mm]

    \pgfmathsetmacro{\xmax}{7.5}
    \pgfmathsetmacro{\ymax}{21.5}

    \pgfmathsetmacro{\lxm}{(2*\ymax)^0.5}

    \pgfmathsetmacro{\d}{0.25}

    % 座標軸
    \draw[semithick] (-\xmax,0) -- (\xmax,0) node[right] {$x$};
    \draw[semithick] (0,-1.5) -- (0,\ymax) node[above] {$y$};
    
    % x軸・y軸の目盛り
    \foreach \x in {-7,-6,...,7} {
        \draw (\x,-\d) -- (\x,\d);
    }
    \foreach \y in {-1,0,...,21} {
        \draw (-\d,\y) -- (\d,\y);
    }
    \foreach \x in {-5,5} {
        \node[below, outer sep=1pt] at (\x,0) {$\x$};
    }
    \foreach \y in {10,15,20} {
        \node[left, outer sep=1pt] at (0,\y) {$\y$};
    }
    \node[right, outer sep=1pt] at (0,5) {$5$};

    % y=0.5x^2のグラフ
    \draw[thick, domain=-\lxm:\lxm, samples=200] plot (\x,{0.5*\x*\x})
        node[above]{$\ell$};

    % 原点
    \node[below left] at (0,0) {$\mathrm{O}$};
    
    % 座標指定
    \coordinate (A) at (-6,18);
    \coordinate (B) at (0,18);
    \coordinate (C) at (-6,0);
    \coordinate (P) at (4.2,8.82);

    % 直線の描画
    \draw[semithick] (A) -- (B) -- (P) -- (C) -- cycle;
    \draw[semithick] (A) -- (P);
    
    % 点の描画
    \fill (A) circle (1.5pt) node[left] {$\mathrm{A}$};
    \fill (P) circle (1.5pt) node[right] {$\mathrm{P}$};
    
    \node[right] at (B) {$\mathrm{B}$};
    \node[above left] at (C) {$\mathrm{C}$};
   
\end{tikzpicture}
\end{document}

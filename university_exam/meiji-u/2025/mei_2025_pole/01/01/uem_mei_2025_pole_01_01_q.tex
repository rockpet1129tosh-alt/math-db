    1個のさいころを続けて4回投げる．
    
    \begin{myenub}
        \item 1回目と2回目は奇数の目，3回目と4回目は偶数の目が出る確率は\\
            \hspace{3\zw}%
            $\displaystyle%
                \myfraca{1}{\myboxb{\hspace{.5\zw}アイ\hspace{.5\zw}}}%
            $\\
            である．
        \item 4回投げたうち少なくとも2回の目が同じ確率は\\
            \hspace{3\zw}%
            $\displaystyle%
                    \myfracc{\myboxb{\hspace{.5\zw}ウエ\hspace{.5\zw}}}%
                    {\myboxb{\hspace{.5\zw}オカ\hspace{.5\zw}}}%
            $\\
            である．
        \item 4回投げたうち異なる2つの数の目がそれぞれ2回出る確率は\\
            \hspace{3\zw}%
            $\displaystyle%
                \myfracc{\myboxb{　キ　}}%
                {\myboxb{\hspace{.5\zw}クケ\hspace{.5\zw}}}%
            $\\
            である．
    \end{myenub}

    \begin{enumerate}
        \item 正の整数 $k$ に対し，
            \begingroup
                \setlength{\abovedisplayskip}{5pt}
                \setlength{\belowdisplayskip}{3pt}
                \setlength{\jot}{0pt}
                \begin{align*}
                    A_k = \int_{\sqrt{k\pi}}^{\sqrt{(k+1)\pi}}\left|\sin(x^2)\right| \, dx
                \end{align*}
            \endgroup
            とおく．次の不等式が成り立つことを示せ．
            \begingroup
                \setlength{\abovedisplayskip}{6pt}
                \setlength{\belowdisplayskip}{4pt}
                \setlength{\jot}{0pt}
                \begin{align*}
                    \myfraca{1}{\sqrt{(k+1)\pi}}\leqq A_k\leqq \myfraca{1}{\sqrt{k\pi}}
                \end{align*}
            \endgroup
        \item 正の整数 $n$ に対し，
            \begingroup
                \setlength{\abovedisplayskip}{5pt}
                \setlength{\belowdisplayskip}{3pt}
                \setlength{\jot}{0pt}
                \begin{align*}
                    B_n = \myfraca{1}{\sqrt{n}}\int_{\sqrt{n\pi}}^{\sqrt{2n\pi}}%
                        \left|\sin(x^2)\right| \, dx
                \end{align*}
            \endgroup
            とおく．極限 $\displaystyle\lim_{n\to\infty}B_n$ を求めよ．
    \end{enumerate}
